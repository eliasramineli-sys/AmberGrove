cmake_minimum_required(VERSION 3.20)
project(AmberGrove VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(amber_engine src/game.cpp src/render.cpp)
target_include_directories(amber_engine PUBLIC src)
if(MSVC)
    target_compile_options(amber_engine PRIVATE /W4 /permissive-)
else()
    target_compile_options(amber_engine PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    add_executable(AmberGrove WIN32 src/main.cpp src/platform_windows.cpp)
    target_compile_definitions(AmberGrove PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(AmberGrove PRIVATE amber_engine gdi32 user32 winmm)
    if(MSVC)
        target_link_options(AmberGrove PRIVATE /ENTRY:mainCRTStartup)
    elseif(MINGW)
        target_link_options(AmberGrove PRIVATE -static -static-libgcc -static-libstdc++)
    endif()
else()
    add_executable(AmberGrove src/main.cpp)
    target_link_libraries(AmberGrove PRIVATE amber_engine)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(amber_tests tests/tests.cpp)
    target_link_libraries(amber_tests PRIVATE amber_engine)
    add_test(NAME game_rules_and_level COMMAND amber_tests)
endif()
